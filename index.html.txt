                            <span class="block text-[10px] text-blue-200">${currentUser.faculty || 'HUFLIT'}</span>
                        </div>
                        <button onclick="handleLogout()" class="bg-red-500/20 hover:bg-red-500/30 text-red-200 border border-red-400/30 text-xs font-semibold px-3 py-2 rounded-xl transition flex items-center">
                            <i class="fa-solid fa-right-from-bracket mr-1.5"></i>Log Out
                        </button>
                    </div>
                `;
            } else {
                container.innerHTML = `
                    <button onclick="openAuthModal(true)" class="text-white hover:text-amber-300 text-sm font-semibold px-3 py-2 transition">
                        Log In
                    </button>
                    <button onclick="openAuthModal(false)" class="bg-huflit-orange hover:bg-orange-600 text-white text-sm font-bold px-4 py-2 rounded-xl shadow transition">
                        Sign Up
                    </button>
                `;
            }
        }

        function openAuthModal(isLogin) {
            authIsLoginMode = isLogin;
            const modal = document.getElementById('auth-modal');
            const title = document.getElementById('modal-title');
            const submitBtn = document.getElementById('auth-submit-btn');
            const toggleBtn = document.getElementById('auth-toggle-btn');
            const nameField = document.getElementById('name-field-container');
            const facultyField = document.getElementById('faculty-field-container');

            if (authIsLoginMode) {
                title.textContent = "HUFLIT Portal Login";
                submitBtn.textContent = "Log In";
                toggleBtn.textContent = "Don't have an account? Sign Up";
                nameField.classList.add('hidden');
                facultyField.classList.add('hidden');
            } else {
                title.textContent = "Create HUFLIT Account";
                submitBtn.textContent = "Sign Up";
                toggleBtn.textContent = "Already have an account? Log In";
                nameField.classList.remove('hidden');
                facultyField.classList.remove('hidden');
            }

            modal.classList.remove('hidden');
        }

        function closeAuthModal() {
            document.getElementById('auth-modal').classList.add('hidden');
        }

        function toggleAuthMode() {
            openAuthModal(!authIsLoginMode);
        }

        function handleAuthSubmit(e) {
            e.preventDefault();
            const email = document.getElementById('auth-email').value.trim();
            const name = document.getElementById('auth-name').value.trim() || email.split('@')[0];
            const faculty = document.getElementById('auth-faculty').value;

            currentUser = {
                email: email,
                name: name,
                faculty: faculty
            };

            localStorage.setItem('huflit_user', JSON.stringify(currentUser));
            renderAuthHeader();
            closeAuthModal();
        }

        function handleLogout() {
            currentUser = null;
            localStorage.removeItem('huflit_user');
            renderAuthHeader();
        }

        function filterResources() {
            const query = document.getElementById('resource-search').value.toLowerCase();
            const category = document.getElementById('category-filter').value;

            const filtered = resourcesData.filter(res => {
                const matchesSearch = res.title.toLowerCase().includes(query) || res.description.toLowerCase().includes(query);
                const matchesCat = category === 'all' || res.category === category;
                return matchesSearch && matchesCat;
            });

            renderResources(filtered);
        }

        // --- PRACTICE QUIZ LOGIC ---
        function renderQuiz() {
            const container = document.getElementById('quiz-container');
            container.innerHTML = quizQuestions.map((q, qIndex) => `
                <div class="space-y-3 p-4 bg-slate-50 rounded-2xl border border-slate-200">
                    <p class="font-bold text-slate-800 text-sm">${qIndex + 1}. ${q.question}</p>
                    <div class="space-y-2">
                        ${q.options.map((opt, optIndex) => `
                            <label class="flex items-center p-3 bg-white border border-slate-200 rounded-xl cursor-pointer hover:bg-blue-50/50 transition">
                                <input type="radio" name="quiz-q-${q.id}" value="${optIndex}" class="text-huflit-blue focus:ring-huflit-blue h-4 w-4">
                                <span class="ml-3 text-sm text-slate-700">${opt}</span>
                            </label>
                        `).join('')}
                    </div>
                </div>
            `).join('');
        }

        function gradeQuiz() {
            let score = 0;
            quizQuestions.forEach(q => {
                const selected = document.querySelector(`input[name="quiz-q-${q.id}"]:checked`);
                if (selected && parseInt(selected.value) === q.correct) {
                    score++;
                }
            });

            const badge = document.getElementById('quiz-score-badge');
            badge.textContent = `Score: ${score}/${quizQuestions.length}`;
            badge.className = score === quizQuestions.length 
                ? "bg-green-100 text-green-800 font-bold text-sm px-3 py-1.5 rounded-xl border border-green-200"
                : "bg-amber-100 text-amber-800 font-bold text-sm px-3 py-1.5 rounded-xl border border-amber-200";
        }

        function resetQuiz() {
            renderQuiz();
            const badge = document.getElementById('quiz-score-badge');
            badge.textContent = `Score: 0/${quizQuestions.length}`;
            badge.className = "bg-blue-50 text-huflit-blue font-bold text-sm px-3 py-1.5 rounded-xl border border-blue-200";
        }

        // --- 5-STAR FEEDBACK & REVIEW LOGIC ---
        function setRating(rating) {
            selectedStarRating = rating;
            document.getElementById('selected-rating').value = rating;
            document.getElementById('rating-error').classList.add('hidden');

            const stars = document.querySelectorAll('#star-picker i');
            stars.forEach((star, idx) => {
                if (idx < rating) {
                    star.className = "fa-solid fa-star text-amber-400";
                } else {
                    star.className = "fa-solid fa-star text-slate-300";
                }
            });
        }

        function loadReviewsFromStorage() {
            const stored = localStorage.getItem('huflit_reviews');
            if (stored) {
                try { defaultReviews = JSON.parse(stored); } catch(e) {}
            }
            renderReviews();
        }

        function renderReviews() {
            const container = document.getElementById('reviews-container');
            if (defaultReviews.length === 0) {
                container.innerHTML = `<p class="text-slate-400 text-sm">No reviews submitted yet.</p>`;
                return;
            }

            // Calculate Avg
            const sum = defaultReviews.reduce((acc, curr) => acc + curr.rating, 0);
            const avg = (sum / defaultReviews.length).toFixed(1);
            document.getElementById('average-rating-badge').textContent = `⭐ Avg: ${avg} / 5.0 (${defaultReviews.length} reviews)`;

            container.innerHTML = defaultReviews.map(rev => `
                <div class="bg-white rounded-2xl p-5 shadow-sm border border-slate-200 space-y-3">
                    <div class="flex justify-between items-start">
                        <div>
                            <h4 class="font-bold text-slate-800 text-sm">${rev.name}</h4>
                            <span class="text-xs text-slate-400">${rev.faculty} • ${rev.date}</span>
                        </div>
                        <div class="text-amber-400 text-xs">
                            ${'★'.repeat(rev.rating)}${'☆'.repeat(5 - rev.rating)}
                        </div>
                    </div>
                    <p class="text-xs text-slate-600 bg-slate-50 p-2.5 rounded-xl">
                        <span class="font-semibold text-huflit-blue">Navigation:</span> ${rev.ease} | 
                        <span class="font-semibold text-huflit-blue">Helpfulness:</span> ${rev.helpfulness}
                    </p>
                    <p class="text-slate-700 text-sm italic">"${rev.comment}"</p>
                </div>
            `).join('');
        }

        function handleFeedbackSubmit(e) {
            e.preventDefault();

            if (selectedStarRating === 0) {
                document.getElementById('rating-error').classList.remove('hidden');
                return;
            }

            const ease = document.getElementById('survey-ease').value;
            const helpfulness = document.getElementById('survey-helpfulness').value;
            const comment = document.getElementById('feedback-comments').value.trim();

            const newReview = {
                name: currentUser ? currentUser.name : "HUFLIT Student",
                faculty: currentUser ? currentUser.faculty : "General",
                rating: selectedStarRating,
                ease: ease,
                helpfulness: helpfulness,
                comment: comment,
                date: new Date().toISOString().split('T')[0]
            };

            defaultReviews.unshift(newReview);
            localStorage.setItem('huflit_reviews', JSON.stringify(defaultReviews));
            renderReviews();

            // Reset Form
            document.getElementById('feedback-form').reset();
            setRating(0);
            alert("Thank you for your feedback! Your review has been published.");
        }
